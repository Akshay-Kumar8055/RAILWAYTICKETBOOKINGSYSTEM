import java.util.ArrayList; import java.util.Arrays; import java.util.List; import java.util.Random; import java.util.Scanner; class Train { int trainNumber; String trainName; List routeStations; String startTime; int totalSeats; int availableSeats; Train(int trainNumber, String trainName, List routeStations, String startTime, int totalSeats) { this.trainNumber = trainNumber; this.trainName = trainName; this.routeStations = new ArrayList<>(); for (String station : routeStations) { this.routeStations.add(station.trim().toUpperCase()); } this.startTime = startTime; this.totalSeats = totalSeats; this.availableSeats = totalSeats; } boolean servesDirectRoute(String from, String to) { int fromIndex = routeStations.indexOf(from.toUpperCase()); this.trainNumber = trainNumber; this.trainName = trainName; this.fromStation = fromStation; this.toStation = toStation; this.passengerName = passengerName; this.age = age; this.gender = gender; } } public class Main { public static void main(String[] args) { List trains = new ArrayList<>();
